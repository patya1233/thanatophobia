local function RemoveESP(object)
    if not object then return end
    for _, obj in ipairs(object:GetDescendants()) do
        if string.find(obj.Name, "Fackhub", 1, true) then
            obj:Destroy()
        end
    end
    if string.find(object.Name, "Fackhub", 1, true) then
        object:Destroy()
    end
end

local function CreateHighlight(object, fillColor, outlineColor, fillTrans, outlineTrans)
    if not object or object:FindFirstChild("FackhubHighlight") then return end
    local highlight = Instance.new("Highlight")
    highlight.Name = "FackhubHighlight"
    highlight.FillColor = fillColor or Color3.fromRGB(255, 255, 255)
    highlight.OutlineColor = outlineColor or Color3.fromRGB(255, 255, 255)
    highlight.FillTransparency = fillTrans or 0.5
    highlight.OutlineTransparency = outlineTrans or 0
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Parent = object
    return highlight
end

local function CreateESPText(object, text, color, offset, size)
    if not object or object:FindFirstChild("FackhubText") then return end
    local billboard = Instance.new("BillboardGui")
    billboard.Name = "FackhubText"
    billboard.Size = size or UDim2.new(0, 150, 0, 40)
    billboard.AlwaysOnTop = true
    billboard.StudsOffset = offset or Vector3.new(0, 3, 0)
    billboard.Parent = object

    local textLabel = Instance.new("TextLabel")
    textLabel.Size = UDim2.new(1, 0, 1, 0)
    textLabel.BackgroundTransparency = 1
    textLabel.Text = text or object.Name
    textLabel.TextColor3 = color or Color3.fromRGB(255, 255, 255)
    textLabel.TextStrokeTransparency = 0
    textLabel.TextScaled = true
    textLabel.Font = Enum.Font.GothamBold
    textLabel.Parent = billboard
    return billboard
end

local function CreateESPBox(object, color, size, transparency)
    if not object then return nil end
    local part = object:IsA("BasePart") and object
        or (object:IsA("Model") and (object:FindFirstChild("HumanoidRootPart") or object.PrimaryPart or object:FindFirstChildWhichIsA("BasePart", true)))
    if not part or part:FindFirstChild("FackhubBox") then return nil end

    local box = Instance.new("BoxHandleAdornment")
    box.Name = "FackhubBox"
    box.Adornee = part
    box.AlwaysOnTop = true
    box.ZIndex = 10
    box.Size = size or part.Size
    box.Color3 = color or Color3.fromRGB(0, 255, 0)
    box.Transparency = transparency or 0.5
    box.Parent = part
    return box
end

local function ApplyESP(target, enabled, text, color, boxSize, useHighlight)
    if not target then return end
    if enabled then
        CreateESPBox(target, color, boxSize)
        CreateESPText(target, text or target.Name, color, Vector3.new(0, 6, 0), UDim2.new(0, 150, 0, 40))
        if useHighlight then
            CreateHighlight(target, color, Color3.fromRGB(255, 255, 255), 0.5, 0)
        end
    else
        RemoveESP(target)
    end
end

local function InteractPrompt(object, delayTime)
    if not object then return end
    local prompt = object:IsA("ProximityPrompt") and object or object:FindFirstChildWhichIsA("ProximityPrompt", true)
    if prompt then
        if delayTime and delayTime > 0 then task.wait(delayTime) end
        prompt.Enabled = true
        prompt.HoldDuration = 0
        prompt.RequiresLineOfSight = false
        fireproximityprompt(prompt)
    end
end

local function InteractClick(object, delayTime)
    if not object then return end
    local cd = object:IsA("ClickDetector") and object or object:FindFirstChildWhichIsA("ClickDetector", true)
    if cd then
        if delayTime and delayTime > 0 then task.wait(delayTime) end
        cd.MaxActivationDistance = 100
        fireclickdetector(cd)
    end
end

local function TeleportTo(targetPosition)
    if not hrp or not targetPosition then return end
    if typeof(targetPosition) == "CFrame" then
        hrp.CFrame = targetPosition
    elseif typeof(targetPosition) == "Vector3" then
        hrp.CFrame = CFrame.new(targetPosition)
    elseif typeof(targetPosition) == "Instance" then
        if targetPosition:IsA("BasePart") then
            hrp.CFrame = targetPosition.CFrame
        elseif targetPosition:IsA("Attachment") then
            hrp.CFrame = CFrame.new(targetPosition.WorldPosition)
        elseif targetPosition:IsA("Model") then
            local root = targetPosition:FindFirstChild("HumanoidRootPart")
                or targetPosition.PrimaryPart
                or targetPosition:FindFirstChildWhichIsA("BasePart", true)
            if root then hrp.CFrame = root.CFrame end
        end
    end
end

local function TeleportToNearestSafeLight(folder)
    if not folder or not hrp then return end
    local nearest, shortest = nil, math.huge
    for _, v in ipairs(folder:GetChildren()) do
        local pos = v:IsA("BasePart") and v.Position or (v:IsA("Model") and v:GetPivot().Position)
        if pos then
            local dist = (hrp.Position - pos).Magnitude
            if dist < shortest then
                shortest = dist
                nearest = v
            end
        end
    end
    if nearest then TeleportTo(nearest) end
end

local savedCFrame = nil

    if hrp then savedCFrame = hrp.CFrame end
end

    if savedCFrame and hrp then hrp.CFrame = savedCFrame end
end

local function GetMonsterRoot(monster)
    if not monster then return nil end
    if monster:IsA("Model") then
        return monster:FindFirstChild("HumanoidRootPart")
            or monster:FindFirstChild("Cube")
            or monster.PrimaryPart
            or monster:FindFirstChildWhichIsA("BasePart", true)
    elseif monster:IsA("BasePart") then
        return monster
    end
    return nil
end

local function IsMonsterNear(monsterPath, target, maxDistance)
    if not monsterPath or not target then return false end
    local targetPosition = (typeof(target) == "Instance" and target:IsA("BasePart") and target.Position)
        or (typeof(target) == "Instance" and target:IsA("Model") and target:GetPivot().Position)
        or (typeof(target) == "Vector3" and target)
    if not targetPosition then return false end

    for _, monster in ipairs(monsterPath:GetChildren()) do
        local root = GetMonsterRoot(monster)
        if root and (root.Position - targetPosition).Magnitude <= maxDistance then
            return true
        end
    end
    return false
end

local FloatConnection = nil

    if FloatConnection then FloatConnection:Disconnect() end
    FloatConnection = RunService.Heartbeat:Connect(function()
        if hrp and hrp.Parent then
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
    end)
end

local function StopFloat()
    if FloatConnection then
        FloatConnection:Disconnect()
        FloatConnection = nil
    end
end

local function SolveWires()
    local obj = workspace:FindFirstChild("MainPuzzle") and workspace.MainPuzzle.Objectives:FindFirstChild("Wires")
    if not obj or obj.Value then return end

    local wc = workspace:FindFirstChild("MapDecor")
        and workspace.MapDecor.MemoryBuilds:FindFirstChild("MinerHut")
        and workspace.MapDecor.MemoryBuilds.MinerHut:FindFirstChild("WireConnect")
    if not wc then return end

    local cList = wc.Connects:GetChildren()
    table.sort(cList, function(a, b) return a.Position.X > b.Position.X end)

    local w1 = wc.Wires:FindFirstChild("Wire1")
    if not w1 or not cList[1] then return end

    local pts = { w1.Position * 2 - cList[1].Position }
    for _, c in ipairs(cList) do table.insert(pts, c.Position) end

    for _ = 1, 2 do
        if obj.Value then break end
        for i = 1, 11 do
            if obj.Value then break end
            local w = wc.Wires:FindFirstChild("Wire" .. i)
            if w and pts[i] and pts[i + 1] then
                local pA, pB = pts[i], pts[i + 1]
                local angle = math.deg(math.atan2(pB.Y - pA.Y, Vector2.new(pB.X - pA.X, pB.Z - pA.Z).Magnitude))
                local target = math.floor((angle / 10) + 0.5) * 10

                TeleportTo(w.CFrame + w.CFrame.LookVector * 3)
                task.wait(0.35)
                for _ = 1, 36 do
                    local cur = math.floor(w.Orientation.Z + 0.5)
                    local diff = math.abs((cur - target) % 180)
                    if diff <= 2 or diff >= 178 or obj.Value then break end
                    InteractPrompt(w, 0)
                    task.wait(0.22)
                end
            end
        end
        task.wait(0.4)
    end
end

local function SolveThreeXThree()
    local obj = workspace:FindFirstChild("MainPuzzle") and workspace.MainPuzzle.Objectives:FindFirstChild("ThreeXThree")
    if not obj or obj.Value then return end

    local tp = workspace:FindFirstChild("TilePuzzle")
    if not tp or not tp:FindFirstChild("MAIN") or not tp:FindFirstChild("Part") then return end

    local masks = {
        [1] = {1, 2, 4}, [2] = {1, 2, 3, 5}, [3] = {2, 3, 6},
        [4] = {1, 4, 5, 7}, [5] = {2, 4, 5, 6, 8}, [6] = {3, 5, 6, 9},
        [7] = {4, 7, 8}, [8] = {5, 7, 8, 9}, [9] = {6, 8, 9},
    }

    for _ = 1, 3 do
        if obj.Value then break end
        TeleportTo(tp.Part.CFrame + Vector3.new(0, 2.5, 0))
        if firetouchinterest then firetouchinterest(hrp, tp.Part, 0) end
        task.wait(0.45)

        local b = {}
        for i = 1, 9 do
            b[i] = (tostring(tp.MAIN["T" .. i].BrickColor) == "Shamrock") and 1 or 0
        end

        for mask = 0, 511 do
            local sim = {table.unpack(b)}
            local moves = {}
            for i = 1, 9 do
                if bit32.band(mask, bit32.lshift(1, i - 1)) ~= 0 then
                    table.insert(moves, i)
                    for _, cell in ipairs(masks[i]) do
                        sim[cell] = 1 - sim[cell]
                    end
                end
            end
            local ok = true
            for i = 1, 9 do
                if sim[i] ~= 1 then ok = false; break end
            end
            if ok then
                for _, m in ipairs(moves) do
                    TeleportTo(tp.Part.CFrame + Vector3.new(0, 2.5, 0))
                    InteractClick(tp.MAIN["T" .. m], 0.18)
                end
                InteractClick(tp.MAIN.ENTER, 0.25)
                task.wait(0.5)
                break
            end
        end
    end
end

local function SolvePillars()
    local obj = workspace:FindFirstChild("MainPuzzle") and workspace.MainPuzzle.Objectives:FindFirstChild("Pillars")
    if not obj or obj.Value then return end

    local ch3 = workspace:FindFirstChild("MapDecor") and workspace.MapDecor.MemoryBuilds:FindFirstChild("Ch3Pillars")
    if not ch3 then return end

    EnsureCheckpointTriggered()

    local pillars = {}
    for _, p in ipairs(ch3:GetChildren()) do
        if p:IsA("Model") then table.insert(pillars, p) end
    end
    local sub = ch3:FindFirstChild("Submit")
    if #pillars < 3 or not sub then return end

    local function clickPillarVerified(pIdx)
        local p = pillars[pIdx]
        local lt = p.LightTrigger
        local sp = lt.SelectPrompt

        for _ = 1, 4 do
            if obj.Value or not sp.Enabled then return false end
            TeleportTo(lt.CFrame + Vector3.new(-3, 2, 0))
            task.wait(0.35)
            fireproximityprompt(sp)

            local t0 = tick()
            local flashed = false
            while (tick() - t0 < 0.45) do
                if tostring(lt.BrickColor) == "Really red" then
                    flashed = true
                    break
                end
                task.wait(0.02)
            end

            if flashed then
                local t1 = tick()
                while tostring(lt.BrickColor) == "Really red" and (tick() - t1 < 1.2) do
                    task.wait(0.02)
                end
                task.wait(0.1)
                return true
            end
        end
        return false
    end

    local safety = 0
    while not obj.Value and safety < 10 do
        safety = safety + 1

        if not pillars[1].LightTrigger.SelectPrompt.Enabled and sub.ProximityPrompt.Enabled then
            TeleportTo(CFrame.new(sub.WorldPosition + Vector3.new(0, 3, 0)))
            task.wait(0.35)
            fireproximityprompt(sub.ProximityPrompt)
            task.wait(0.1)
        elseif pillars[1].LightTrigger.SelectPrompt.Enabled then
            for _ = 1, 8 do
                if not pillars[1].LightTrigger.SelectPrompt.Enabled or obj.Value then break end
                clickPillarVerified(1)
            end
        end

        if obj.Value then break end

        local seq = {}
        local lastRed = ""
        local tRec = tick()
        while not pillars[1].LightTrigger.SelectPrompt.Enabled and not obj.Value and (tick() - tRec < 15) do
            local redList = {}
            for i, p in ipairs(pillars) do
                if tostring(p.LightTrigger.BrickColor) == "Really red" then
                    table.insert(redList, tostring(i))
                end
            end
            local cur = table.concat(redList, "+")
            if cur ~= lastRed then
                if #redList == 1 then
                    table.insert(seq, tonumber(redList[1]))
                end
                lastRed = cur
            end
            task.wait(0.02)
        end

        if obj.Value then break end

        if #seq > 0 then
            task.wait(0.2)
            for _, pIdx in ipairs(seq) do
                if obj.Value or not pillars[1].LightTrigger.SelectPrompt.Enabled then break end
                clickPillarVerified(pIdx)
            end
            task.wait(0.8)
        end
    end
end

local function IsPhase1BossAttacking()
    local bm1 = workspace:FindFirstChild("Boss") and workspace.Boss:FindFirstChild("BossModel1")
    local hum = bm1 and bm1:FindFirstChildOfClass("Humanoid")
    local animator = hum and hum:FindFirstChildOfClass("Animator")
    if not animator then return false, nil end

    for _, tr in ipairs(animator:GetPlayingAnimationTracks()) do
        if tr.Priority == Enum.AnimationPriority.Action4 and tr.IsPlaying and tr.TimePosition < (tr.Length - 0.15) then
            return true, tr
        end
    end
    return false, nil
end

local isHealingCrystal = false
local function ConsumeNearestCrystal()

    local pb = workspace:FindFirstChild("PureBloom")
    if not pb or not hrp then return false end
    for _, c in ipairs(pb:GetChildren()) do

-- ฟังก์ชันโจมตีบอสที่ใช้ได้ทั้ง Phase 1 (Katana) และ Phase 2 (WhiteFlameTorch Aimbot)
local function PerformBossAttack()
    if isSwinging or not character or not hrp or isHealingCrystal or game:HasTag("InCutscene") then return end
    local hum = character:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end

    local bossFolder = workspace:FindFirstChild("Boss")
    local cfg = bossFolder and bossFolder:FindFirstChild("Configuration")
    if not cfg then return end

    -- Phase 1: ใช้ดาบ Katana ฟันเมื่ออยู่ใกล้บอส (ระยะ < 16 studs) และบอสไม่ได้ออกท่าโจมตี
    if cfg.P1Health.Value > 0 and bossFolder:FindFirstChild("BossModel1") then
        if IsPhase1BossAttacking() then return end
        local bRoot = bossFolder.BossModel1:FindFirstChild("HumanoidRootPart")
        if not bRoot or (hrp.Position - bRoot.Position).Magnitude > 16 then return end

        local katana = character:FindFirstChild("Katana") or (player:FindFirstChild("Backpack") and player.Backpack:FindFirstChild("Katana"))
        if katana and katana:FindFirstChild("Handle") and katana:FindFirstChild("SwordSwingCFrame") then
            isSwinging = true
            if katana.Parent ~= character then
                hum:EquipTool(katana)
                task.wait(0.1)
            end
            katana:Activate()
            task.wait(0.1)
            local tEnd = os.clock() + 0.55
            while os.clock() < tEnd and cfg.P1Health.Value > 0 and (killAuraActive or autoFarmBossActive) do
                swingCounter = swingCounter + 1
                katana.SwordSwingCFrame:FireServer(swingCounter, katana.Handle.CFrame, workspace:GetServerTimeNow())
                RunService.RenderStepped:Wait()
            end
            task.wait(0.5)
            isSwinging = false
        end

    -- Phase 2: ใช้คบเพลิง WhiteFlameTorch ล็อกเป้าปาใส่ BossModel2.Hitboxes.BodyBox อัตโนมัติ
    elseif cfg.P1Health.Value <= 0 and cfg.P2Health.Value > 0 and bossFolder:FindFirstChild("BossModel2") then
        if os.clock() - lastTorchThrow < 0.25 then return end
        local bm2 = bossFolder.BossModel2
        local bodyBox = bm2:FindFirstChild("Hitboxes") and bm2.Hitboxes:FindFirstChild("BodyBox")
        local torch = character:FindFirstChild("WhiteFlameTorch") or (player:FindFirstChild("Backpack") and player.Backpack:FindFirstChild("WhiteFlameTorch"))
        if bodyBox and torch then
            lastTorchThrow = os.clock()
            if torch.Parent ~= character then
                hum:EquipTool(torch)
            end
            if torch:FindFirstChild("MouseLocation") then
                torch.MouseLocation.OnClientInvoke = function()
                    return hrp.Position, bodyBox.Position
                end
            end
            if torch:FindFirstChild("TakeControl") then
                torch.TakeControl:FireServer("take_control")
                torch:Activate()
                torch.TakeControl:FireServer("throw")
            end
        end
    end
end

-- ลูปหลักรวมการทำงานของทั้ง Part 2 และ Part 3
-- ==========================================
RunService.Heartbeat:Connect(function()
    if not character or not hrp then return end
    local hum = character:FindFirstChildOfClass("Humanoid")

    -- [Part 2] Infinite Stamina
    if infStaminaActive then
        local stam = character:FindFirstChild("Stamina")
        if stam and stam.Value < 100 then stam.Value = 100 end
    end

    -- [Part 2] Auto Calm Down
    if autoCalmActive then
        local ff = character:FindFirstChild("FearFolder")
        local fv = ff and ff:FindFirstChild("FearVal")
        if fv and fv.Value > 30 and ReplicatedStorage:FindFirstChild("CalmDown") then
            task.spawn(function()
                pcall(function() ReplicatedStorage.CalmDown:InvokeServer("CalmDown", 40) end)
            end)
        end
    end

    -- [Part 2] Auto SafeZone TP
    if autoSafeTPActive and workspace:FindFirstChild("Monsters") and workspace:FindFirstChild("SafeZones") then
        local saito = workspace.Monsters:FindFirstChild("DistortedSaito")
        local chasing = saito and saito:FindFirstChild("Values") and saito.Values:FindFirstChild("Chasing") and saito.Values.Chasing.Value
        if chasing or IsMonsterNear(workspace.Monsters, hrp, 35) then
            TeleportToNearestSafeLight(workspace.SafeZones)
        end
    end

    -- [Part 2] Anti-Fear Blur
    if antiEffectsActive then
        local blur = game.Lighting:FindFirstChild("FearStaticBlur")
        if blur then blur.Size = 0 end
        local pg = player:FindFirstChild("PlayerGui")
        if pg then
            if pg:FindFirstChild("StaticUI") and pg.StaticUI:FindFirstChild("staticframe") then
                pg.StaticUI.staticframe.ImageTransparency = 1
            end
            if pg:FindFirstChild("ChaseScreen") then
                pg.ChaseScreen.Enabled = false
            end
        end
    end

    -- [Part 2] อัปเดต ESP EscapeDoor อัตโนมัติทันทีที่ประตูถูกเสกขึ้นมา
    if espEscapeDoorActive and workspace:FindFirstChild("EscapeDoor") then
        ApplyESP(workspace.EscapeDoor, true, "EXIT DOOR", Color3.fromRGB(0, 255, 0), Vector3.new(6, 10, 6), true)
    end

    -- [Part 3] Auto Shockwave & Anti-Obsession
    if autoShockwaveBypass then
        EnableSkillCheckBypass()
        if hum and hum.WalkSpeed < 18 then hum.WalkSpeed = 18 end
    end

    if antiObsessionActive then
        pcall(function() RunService:UnbindFromRenderStep("cameralol") end)
        if hum and hum.WalkSpeed < 18 then hum.WalkSpeed = 18 end
    end

    -- [Part 3] Infinite Dash (ลบ DashKnockback)
    if infDashActive then
        local kb = hrp:FindFirstChild("DashKnockback")
        if kb then kb:Destroy() end
    end

    -- [Part 3] Auto Heal Crystal เมื่อเลือดต่ำกว่า 75
    if autoHealCrystalActive and not isHealingCrystal and not game:HasTag("InCutscene") and (os.clock() - lastHealTick > 2.5) then
        if hum and hum.Health > 0 and hum.Health <= 75 then
            lastHealTick = os.clock()
            task.spawn(ConsumeNearestCrystal)
        end
    end

    -- [Part 3] Boss Fight Logic
    local bossFolder = workspace:FindFirstChild("Boss")
    if bossFolder then
        if espBossAttacksActive and bossFolder:FindFirstChild("BossAttacks") then
            for _, a in ipairs(bossFolder.BossAttacks:GetChildren()) do
                ApplyESP(a, true, a.Name, Color3.fromRGB(255, 120, 0))
            end
        end

        if game:HasTag("InCutscene") or isHealingCrystal or not hum or hum.Health <= 0 then
            return
        end

        local cfg = bossFolder:FindFirstChild("Configuration")
        if not cfg then return end

        -- 1. Auto Block เฉพาะตอนบอสเฟส 1 เริ่มออกท่าโจมตี (Action4) และอยู่ในระยะใกล้ (< 24 studs)
        if autoBlockActive and cfg.P1Health.Value > 0 and bossFolder:FindFirstChild("BossModel1") then
            local bRoot = bossFolder.BossModel1:FindFirstChild("HumanoidRootPart")
            local attacking, tr = IsPhase1BossAttacking()
            if attacking and tr and not blockedTracks[tr] and bRoot and (hrp.Position - bRoot.Position).Magnitude < 24 then
                blockedTracks[tr] = true
                local katana = character:FindFirstChild("Katana")
                if katana and katana:FindFirstChild("BlockRequest") then
                    task.spawn(function()
                        pcall(function() katana.BlockRequest:InvokeServer() end)
                    end)
                end
            end
        end

        -- 2. Auto Farm Boss P1 & P2 (ลอยฟัน + หลบท่าโจมตีอัตโนมัติ)
        if autoFarmBossActive then
            if cfg.P1Health.Value > 0 and bossFolder:FindFirstChild("BossModel1") then
                local bRoot = bossFolder.BossModel1:FindFirstChild("HumanoidRootPart")
                if bRoot then
                    if IsPhase1BossAttacking() then
                        -- บอสกำลังออกท่าโจมตี -> ดีดตัวหลบไปมุมสูงด้านหลังบอส (ไม่โดนทั้งดาบและคลื่นเสา)
                        hrp.CFrame = bRoot.CFrame * CFrame.new(0, 26, 22)
                    else
                        -- บอสไม่ได้ออกท่า -> ลอยเหนือหัวบอสที่ Y+12.5 (อยู่ใน BodyBox แต่สูงกว่าดาบบอส KatanaBox)
                        hrp.CFrame = CFrame.new(bRoot.Position + Vector3.new(0, 12.5, 0))
                    end
                    hrp.AssemblyLinearVelocity = Vector3.zero
                    hrp.AssemblyAngularVelocity = Vector3.zero
                end
            elseif cfg.P1Health.Value <= 0 and cfg.P2Health.Value > 0 and bossFolder:FindFirstChild("BossModel2") then
                local bm2 = bossFolder.BossModel2
                local bodyBox = bm2:FindFirstChild("Hitboxes") and bm2.Hitboxes:FindFirstChild("BodyBox")
                if bodyBox then
                    -- เฟส 2: บินวนเป็นวงกลมรัศมี 55 สูง +28 รอบบอส เพื่อหลบหนวดและหินบนพื้น 100%
                    local angle = os.clock() * 2.5
                    local offset = Vector3.new(math.cos(angle) * 55, 28, math.sin(angle) * 55)
                    hrp.CFrame = CFrame.new(bodyBox.Position + offset, bodyBox.Position)
                    hrp.AssemblyLinearVelocity = Vector3.zero
                    hrp.AssemblyAngularVelocity = Vector3.zero
                end
            end
        end

        -- 3. โจมตีอัตโนมัติ (Kill Aura / Torch Aimbot)
        if autoFarmBossActive or killAuraActive then
            task.spawn(PerformBossAttack)
        end

        -- 4. Auto Execute Saito (เมื่อเลือดเฟส 1 หมดและปุ่ม Kill Him เปิดใช้งาน)
        if (autoExecuteSaitoActive or autoFarmBossActive) and cfg.P1Health.Value <= 0 and bossFolder:FindFirstChild("BossModel1") then
            local pp = bossFolder.BossModel1:FindFirstChildWhichIsA("ProximityPrompt", true)
            if pp and pp.Enabled then
                local bRoot = bossFolder.BossModel1:FindFirstChild("HumanoidRootPart")
                if bRoot then
                    hrp.CFrame = bRoot.CFrame * CFrame.new(0, 0, -4)
                end
                InteractPrompt(pp, 0)
            end
        end
    end
end)
